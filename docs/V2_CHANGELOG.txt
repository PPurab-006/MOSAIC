+def evaluate_predictions(y_true: np.ndarray, probs: np.ndarray, threshold: float) -> Dict:
+    """Computes full classification and ranking metrics."""
+    preds = (probs >= threshold).astype(int)
+    n_pos = int(np.sum(y_true))
+    n_tot = len(y_true)
+    pos_rate = float(n_pos / n_tot * 100.0) if n_tot > 0 else 0.0
+
+    if len(np.unique(y_true)) > 1:
+        auroc = float(roc_auc_score(y_true, probs))
+        auprc = float(average_precision_score(y_true, probs))
+    else:
+        auroc = float("nan")
+        auprc = float("nan")
+
+    f1 = float(f1_score(y_true, preds, zero_division=0))
+    prec = float(precision_score(y_true, preds, zero_division=0))
+    rec = float(recall_score(y_true, preds, zero_division=0))
+
+    if len(np.unique(y_true)) > 1:
+        tn, fp, fn, tp = confusion_matrix(y_true, preds, labels=[0, 1]).ravel()
+        fpr = float(fp / (fp + tn)) if (fp + tn) > 0 else 0.0
+    else:
+        tn = int(np.sum((y_true == 0) & (preds == 0)))
+        fp = int(np.sum((y_true == 0) & (preds == 1)))
+        fn = int(np.sum((y_true == 1) & (preds == 0)))
+        tp = int(np.sum((y_true == 1) & (preds == 1)))
+        fpr = float("nan")
+
+    return {
+        "samples": n_tot,
+        "positives": n_pos,
+        "positive_rate_pct": round(pos_rate, 2),
+        "threshold": round(threshold, 3),
+        "auroc": round(auroc, 4),
+        "auprc": round(auprc, 4),
+        "f1": round(f1, 4),
+        "precision": round(prec, 4),
+        "recall": round(rec, 4),
+        "fpr": round(fpr, 4) if not np.isnan(fpr) else float("nan"),
+        "tn": int(tn),
+        "fp": int(fp),
+        "fn": int(fn),
+        "tp": int(tp),
+    }
